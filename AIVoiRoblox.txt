function Toggle(text, default, callback)
    local t = Instance.new("Frame", Scroll)
    t.Size = UDim2.new(1, 0, 0, 38)
    t.BackgroundColor3 = Color3.fromRGB(30, 30, 35)
    Instance.new("UICorner", t).CornerRadius = UDim.new(0, 8)
    
    local l = Instance.new("TextLabel", t)
    l.Size = UDim2.new(0, 160, 1, 0)
    l.Position = UDim2.new(0, 10, 0, 0)
    l.Text = text
    l.TextColor3 = Color3.fromRGB(255, 255, 255)
    l.TextSize = 13
    l.Font = Enum.Font.GothamMedium
    l.TextXAlignment = Enum.TextXAlignment.Left
    l.BackgroundTransparency = 1
    
    local b = Instance.new("Frame", t)
    b.Size = UDim2.new(0, 42, 0, 22)
    b.Position = UDim2.new(1, -52, 0.5, -11)
    b.BackgroundColor3 = default and Color3.fromRGB(0, 255, 255) or Color3.fromRGB(60, 60, 65)
    Instance.new("UICorner", b).CornerRadius = UDim.new(1, 0)
    
    local d = Instance.new("Frame", b)
    d.Size = UDim2.new(0, 16, 0, 16)
    d.Position = default and UDim2.new(1, -19, 0.5, -8) or UDim2.new(0, 3, 0.5, -8)
    d.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    Instance.new("UICorner", d).CornerRadius = UDim.new(1, 0)
    
    local on = default
    b.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            on = not on
            TS:Create(b, TweenInfo.new(0.2), {BackgroundColor3 = on and Color3.fromRGB(0, 255, 255) or Color3.fromRGB(60, 60, 65)}):Play()
            TS:Create(d, TweenInfo.new(0.2), {Position = on and UDim2.new(1, -19, 0.5, -8) or UDim2.new(0, 3, 0.5, -8)}):Play()
            callback(on)
        end
    end)
    return t
end

function Slider(text, min, max, default, callback)
    local s = Instance.new("Frame", Scroll)
    s.Size = UDim2.new(1, 0, 0, 55)
    s.BackgroundColor3 = Color3.fromRGB(30, 30, 35)
    Instance.new("UICorner", s).CornerRadius = UDim.new(0, 8)
    
    local l = Instance.new("TextLabel", s)
    l.Size = UDim2.new(1, -20, 0, 20)
    l.Position = UDim2.new(0, 10, 0, 3)
    l.Text = text .. ": " .. default
    l.TextColor3 = Color3.fromRGB(255, 255, 255)
    l.TextSize = 12
    l.Font = Enum.Font.GothamMedium
    l.TextXAlignment = Enum.TextXAlignment.Left
    l.BackgroundTransparency = 1
    
    local bar = Instance.new("Frame", s)
    bar.Size = UDim2.new(1, -20, 0, 6)
    bar.Position = UDim2.new(0, 10, 0, 30)
    bar.BackgroundColor3 = Color3.fromRGB(50, 50, 55)
    Instance.new("UICorner", bar).CornerRadius = UDim.new(0, 3)
    
    local fill = Instance.new("Frame", bar)
    fill.Size = UDim2.new((default - min) / (max - min), 0, 1, 0)
    fill.BackgroundColor3 = Color3.fromRGB(0, 255, 255)
    Instance.new("UICorner", fill).CornerRadius = UDim.new(0, 3)
    
    local function update(i)
        local rel = math.clamp((i.Position.X - bar.AbsolutePosition.X) / bar.AbsoluteSize.X, 0, 1)
        local val = math.floor((min + (max - min) * rel) * 100) / 100
        l.Text = text .. ": " .. val
        fill.Size = UDim2.new(rel, 0, 1, 0)
        callback(val)
    end
    
    bar.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            update(i)
            local con
            con = UIS.InputChanged:Connect(function(c)
                if c.UserInputType == Enum.UserInputType.MouseMovement or c.UserInputType == Enum.UserInputType.Touch then
                    update(c)
                end
            end)
            UIS.InputEnded:Connect(function(e)
                if e.UserInputType == Enum.UserInputType.MouseButton1 or e.UserInputType == Enum.UserInputType.Touch then
                    con:Disconnect()
                end
            end)
        end
    end)
    return s
end

-- ===== FEATURES =====
local flying, noclip = false, false
local flySpeed = 50

Section("🚀 Movement")

Toggle("Enable Fly", false, function(v)
    flying = v
    local char = LP.Character
    if v and char and char:FindFirstChild("HumanoidRootPart") then
        local gyro = Instance.new("BodyGyro", char.HumanoidRootPart)
        gyro.P = 9e4
        gyro.MaxTorque = Vector3.new(9e4, 9e4, 9e4)
        local vel = Instance.new("BodyVelocity", char.HumanoidRootPart)
        vel.MaxForce = Vector3.new(9e4, 9e4, 9e4)
        
        local con
        con = RS.RenderStepped:Connect(function()
            if not flying or not char.Parent then
                gyro:Destroy()
                vel:Destroy()
                con:Disconnect()
                return
            end
            gyro.CFrame = Cam.CFrame
            local move = Vector3.new()
            if UIS:IsKeyDown(Enum.KeyCode.W) then move += Cam.CFrame.LookVector end
            if UIS:IsKeyDown(Enum.KeyCode.S) then move -= Cam.CFrame.LookVector end
            if UIS:IsKeyDown(Enum.KeyCode.A) then move -= Cam.CFrame.RightVector end
            if UIS:IsKeyDown(Enum.KeyCode.D) then move += Cam.CFrame.RightVector end
            if UIS:IsKeyDown(Enum.KeyCode.Space) then move += Vector3.new(0, 1, 0) end
            if UIS:IsKeyDown(Enum.KeyCode.LeftControl) then move -= Vector3.new(0, 1, 0) end
            if move.Magnitude > 0 then vel.Velocity = move.Unit * flySpeed end
        end)
    end
end)

Slider("Fly Speed", 10, 500, 50, function(v) flySpeed = v end)

Toggle("Noclip", false, function(v)
    noclip = v
    if v then
        local con
        con = RS.Stepped:Connect(function()
            if not noclip then con:Disconnect() return end
            if LP.Character then
                for _, p in ipairs(LP.Character:GetDescendants()) do
                    if p:IsA("BasePart") then p.CanCollide = false end
                end
            end
        end)
    end
end)
